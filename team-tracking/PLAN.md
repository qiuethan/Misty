# team-tracking Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the team-tracking source-of-truth service — a Python HTTP API over four base tables (people, teams, role_kinds, team_memberships) that all downstream UTMIST systems will consume.

**Architecture:** Three-layer split — stable `contracts/` (Pydantic types + `StorageAdapter` Protocol), FastAPI routes in `src/api/` depending only on the contracts, and swappable concrete storage in `src/storage/` (in-memory adapter for tests, Postgres adapter for production). Internal boundaries expressed as `typing.Protocol` interfaces. External contract is OpenAPI auto-generated by FastAPI. No consumers run in-process; every downstream system talks HTTP.

**Tech Stack:** Python 3.11+, uv (package manager), FastAPI, Pydantic v2, SQLAlchemy Core 2.0, psycopg 3, Alembic (migrations), pytest + httpx (tests), Docker Compose (local Postgres).

**Spec reference:** `team-tracking/DESIGN.md`

- **`contracts/`** — pure Python, no framework imports (no FastAPI, no SQLAlchemy). Only `pydantic`, `typing`, `uuid`, `datetime`.

```python
from datetime import date
from typing import Protocol
from uuid import UUID

from contracts.types import (
    Person,
    PersonCreate,
    PersonUpdate,
    RoleKind,
    Team,
    TeamCreate,
    TeamMembership,
    TeamMembershipCreate,
    TeamMembershipUpdate,
    TeamUpdate,
)


class StorageAdapter(Protocol):
    """Stable internal contract between the API and the persistence layer.

    Any concrete adapter (Postgres, SQLite, in-memory, ...) implements these
    methods with identical semantics. Consumers of the API never see this
    interface — only the API layer does.

    Conventions:
    - All methods that write also stamp created_by/updated_by using the actor arg.
    - `list_*` methods that take filters return every matching record; pagination
      is not implemented in v1 (add later if the roster grows past ~500 people).
    - `end_membership` is a semantic helper: sets ended_at without deleting the row.
    """

    # People
    def create_person(self, payload: PersonCreate, *, actor: str) -> Person: ...
    def get_person(self, person_id: UUID) -> Person | None: ...
    def get_person_by_email(self, primary_email: str) -> Person | None: ...
    def list_people(self, *, active_only: bool = False) -> list[Person]: ...
    def update_person(
        self, person_id: UUID, payload: PersonUpdate, *, actor: str
    ) -> Person | None: ...

    # Teams
    def create_team(self, payload: TeamCreate, *, actor: str) -> Team: ...
    def get_team(self, team_id: UUID) -> Team | None: ...
    def get_team_by_slug(self, slug: str) -> Team | None: ...
    def list_teams(self, *, active_only: bool = False) -> list[Team]: ...
    def update_team(
        self, team_id: UUID, payload: TeamUpdate, *, actor: str
    ) -> Team | None: ...

    # Role kinds
    def list_role_kinds(self, *, active_only: bool = False) -> list[RoleKind]: ...
    def get_role_kind(self, role_kind_id: str) -> RoleKind | None: ...

    # Team memberships
    def create_membership(
        self, payload: TeamMembershipCreate, *, actor: str
    ) -> TeamMembership: ...
    def get_membership(self, membership_id: UUID) -> TeamMembership | None: ...
    def list_memberships(
        self,
        *,
        team_id: UUID | None = None,
        person_id: UUID | None = None,
        active_only: bool = False,
        as_of: date | None = None,
        is_team_admin: bool | None = None,
    ) -> list[TeamMembership]: ...
    def update_membership(
        self, membership_id: UUID, payload: TeamMembershipUpdate, *, actor: str
    ) -> TeamMembership | None: ...
    def end_membership(
        self, membership_id: UUID, ended_at: date, *, actor: str
    ) -> TeamMembership | None: ...
```

from datetime import date, datetime
from uuid import UUID, uuid4

from contracts.types import (
    Person,
    PersonCreate,
    PersonUpdate,
    RoleKind,
    Team,
    TeamCreate,
    TeamMembership,
    TeamMembershipCreate,
    TeamMembershipUpdate,
    TeamUpdate,
)


def _now() -> datetime:
    return datetime.utcnow()


class InMemoryStorageAdapter:
    """In-process storage adapter used in tests and for quick prototyping.

    Not thread-safe. Not persistent. Enforces the same semantic invariants as
    the Postgres adapter (email uniqueness, slug uniqueness, no schema surprises).
    """

    def __init__(self, seed_role_kinds: list[RoleKind] | None = None) -> None:
        self._people: dict[UUID, Person] = {}
        self._teams: dict[UUID, Team] = {}
        self._role_kinds: dict[str, RoleKind] = {
            rk.id: rk for rk in (seed_role_kinds or [])
        }
        self._memberships: dict[UUID, TeamMembership] = {}

    # --- People ---

    def create_person(self, payload: PersonCreate, *, actor: str) -> Person:
        email = payload.primary_email
        if any(p.primary_email == email for p in self._people.values()):
            raise ValueError(f"primary_email already exists: {email}")
        now = _now()
        p = Person(
            id=uuid4(),
            display_name=payload.display_name,
            primary_email=email,
            active=True,
            created_at=now,
            updated_at=now,
            created_by=actor,
            updated_by=actor,
        )
        self._people[p.id] = p
        return p

    def get_person(self, person_id: UUID) -> Person | None:
        return self._people.get(person_id)

    def get_person_by_email(self, primary_email: str) -> Person | None:
        target = primary_email.strip().lower()

        return None

        self, person_id: UUID, payload: PersonUpdate, *, actor: str

        data = existing.model_dump()
        patch = payload.model_dump(exclude_unset=True)
        if "primary_email" in patch and patch["primary_email"] != existing.primary_email:
            new_email = patch["primary_email"]
            if any(
                p.primary_email == new_email and p.id != person_id
                for p in self._people.values()
            ):
                raise ValueError(f"primary_email already exists: {new_email}")
        data.update(patch)
        data["updated_at"] = _now()
        data["updated_by"] = actor
        updated = Person(**data)
        self._people[person_id] = updated
        return updated

            person_id=payload.person_id,
            team_id=payload.team_id,
            role_kind_id=payload.role_kind_id,

            started_at=payload.started_at or date.today(),
            ended_at=payload.ended_at,

            updated_at=now,

        is_team_admin: bool | None = None,
    ) -> list[TeamMembership]:
        results = list(self._memberships.values())
        if team_id is not None:
            results = [m for m in results if m.team_id == team_id]
        if person_id is not None:

        if active_only:
            results = [m for m in results if m.ended_at is None]

                m for m in results
                if m.started_at <= as_of and (m.ended_at is None or m.ended_at > as_of)
            ]
        if is_team_admin is not None:
            results = [m for m in results if m.is_team_admin is is_team_admin]

        data.update(patch)
        data["updated_at"] = _now()
        data["updated_by"] = actor
        updated = TeamMembership(**data)
        self._memberships[membership_id] = updated
        return updated

        self, membership_id: UUID, ended_at: date, *, actor: str

            conditions.append(team_memberships.c.is_team_admin.is_(is_team_admin))
        if conditions:
            stmt = stmt.where(and_(*conditions))
        with self._engine.connect() as conn:
            rows = conn.execute(stmt).all()
        return [_membership_row_to_model(r) for r in rows]

    def update_membership(
        self, membership_id: UUID, payload: TeamMembershipUpdate, *, actor: str

        if not patch:

            ).one_or_none()

def require_api_key(
    x_api_key: str | None = Header(default=None, alias="X-API-Key"),
) -> str:
    """Verifies the API key. Raises 401 if invalid. Returns 'api'."""
    if x_api_key is None or x_api_key != get_settings().api_key:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or missing X-API-Key",
        )
    return "api"


def get_actor(
    _: str = Depends(require_api_key),
    x_actor: str | None = Header(default=None, alias="X-Actor"),
) -> str:
    """Composite dependency: enforce API key AND return the actor identifier.

    Prefer explicit X-Actor header (e.g. 'discord-bot', 'sync-job'); fall back to 'api'.
    The returned string is used for created_by/updated_by on writes. Import this in
    routers instead of redefining per-file.
    """
    return x_actor or "api"

- **No pagination on list endpoints.** Add when the roster grows past ~500 rows.
